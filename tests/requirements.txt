# Python packages for the robot test suite.
#
# Single source of truth: create-dev-container.sh installs from this file, and
# README.md documents it.
#
#   cd tests
#   uv venv --python=/usr/bin/python3 robotframework   # first time only
#   uv pip install --python robotframework/bin/python -U -r requirements.txt
#
# Deliberately NOT pinned: the point here is to update the venv. Freeze the
# versions (uv pip freeze) if a reproducible environment is ever needed.
#
# Note that `uv pip install` never removes anything; `uv pip sync` is what
# actually aligns an existing venv on this list.
#
# Not listed because nothing imports them: boto3, py-cpuinfo, cython and
# gitpython were the dependencies of the removed S3 benchmark. The
# `opentelemetry` import resolves to resources/opentelemetry, generated by
# init-proto.sh, and needs no package.

# Robot Framework itself, and the libraries resources/import.resource declares.
robotframework
robotframework-databaselibrary
robotframework-examples
robotframework-httpctrl
robotframework-requests

# How Engine.py and Broker.py talk to the daemons; grpcio-tools is what
# init-proto.sh generates the stubs with (it pulls protobuf in).
grpcio
grpcio-tools

# The driver DatabaseLibrary and Common.py use.
pymysql

# Watches processes (Common.py, Broker.py).
psutil

# Used by the vault tests.
pyjwt

# Formatters: autopep8 for the Python under resources/, robotframework-tidy for
# the .robot files. See the "Coding style" section of README.md.
autopep8
robotframework-tidy
